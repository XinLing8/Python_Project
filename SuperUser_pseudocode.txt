#CreateSuperUser_data
FUNCTION CreateSuperUser_data
	DEFINE a list named SuperUser_data with single string 'superuser,superpass'
	OPEN file 'SuperUser.txt' in write mode
		FOR each line in SuperUser_data
			WRITE the item to the file followed by a newline
		CLOSE file
		PRINT "Super User data file created with initial data."

#SuperUser_login
FUNCTION SuperUser_login
	SET max_attempts TO 3
	SET attempts TO 0
	
	WHILE attempts < max_attempts
		PROMPT user for Super User ID
		READ superuser_id
		PROMPT user for password
		READ password
		
		SET valid_login TO False
		
		TRY
			OPEN file 'SuperUser.txt' in read mode
				FOR each line in file
					SPLIT line into stored_superuser_id AND store_password
					IF stored_superuser_id matches superuser_id AND stored_password matches password
						SET valid_login TO True
						BREAK loop
		EXCEPT FileNotFoundError
			PRINT "User data file not found."
			RETURN
			
		IF valid_login
			PRINT "Login successful."
			CALL SuperUser_menu with superuser_id
			RETURN
		ELSE
			INCREMENT attempts
			PRINT number of attempts left
			
	PRINT "Too many failed attempts. Access denied."
	CALL reset_superuser_password with superuser_id

#Verify_New_Users	
FUNCTION Verify_New_Users
	TRY
		OPEN 'users.txt' in read mode
			READ all lines into users
	EXCEPT FileNotFoundError
		RETURN
		
	SET pending_users TO users where status is "pending"
	
	IF pending_users is empty
        PRINT "No pending users to verify."
        RETURN

    FOR each user in pending_users with index i
        PRINT user details

    LOG action with superuser_id and "Verify New Users"

    WHILE True
        PROMPT user for decision (Y/N)
        IF decision is valid
            BREAK loop
        PRINT invalid choice message

    IF decision is 'Y'
        CALL Add_User
    ELSE
        PRINT "No users were processed."

#Add_User		
FUNCTION Add_User
    LOG action with superuser_id and "Add User"
    TRY
        OPEN 'users.txt' in read mode
            READ all lines into users
    EXCEPT FileNotFoundError
        RETURN

    SET pending_users TO users where status is "pending"

    IF pending_users is empty
        PRINT "No pending users to add."
        RETURN

    FOR each user in pending_users with index i
        PRINT user details

        WHILE True
            PROMPT user for decision (A/R)
            IF decision is valid
                BREAK loop
            PRINT invalid choice message

        IF decision is 'A'
            WHILE True
                PROMPT user for role (S/A)
                IF role is valid
                    BREAK loop
                PRINT invalid role message

            UPDATE user status to "approved" and save user
            CALL save_user_to_file
            PRINT success message
        ELSE
            UPDATE user status to "rejected"
            PRINT rejection message

    TRY
        OPEN 'users.txt' in write mode
            WRITE updated users
    EXCEPT Exception as e
        PRINT error message

#save_user_to_file		
FUNCTION save_user_to_file(username, password, ic, role)
    IF role is 'staff'
        SET file_path TO 'staff.txt'
    ELSE IF role is 'admin'
        SET file_path TO 'admin.txt'
	ELSE IF role is 'customer'
        SET file_path TO 'customers.txt'
    ELSE
        RETURN

    TRY
        OPEN file_path in append mode
            WRITE user details
    EXCEPT Exception as e
        PRINT error message
		
#Modify_User_Details
FUNCTION Modify_User_Details
    LOG action with superuser_id and "Modify User Details"
    TRY
        OPEN 'users.txt' in read mode
            READ all lines into users
    EXCEPT FileNotFoundError
        RETURN

    SET user_dict TO dictionary of username to user details

    IF user_dict is empty
        PRINT "No valid users found."
        RETURN

    FOR each user in user_dict
        PRINT user details

    PROMPT user for selection
    IF selection is valid
        UPDATE user details
        WRITE updated user details to 'users.txt'
        PRINT success message
    ELSE
        PRINT invalid selection message
